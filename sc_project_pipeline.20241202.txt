# call ATAC-seq peaks on bulk scATAC-seq data - only run on early embryo data (exp031/032/042/043)
### this new step is needed to prevent the extension of the 3' UTR of a transcript beyond an accessible site (AS)
### AS are often sites of active transcription; therefore, an AS in the 3' UTR might be associated with RNA-seq reads
### which would erroneously be assigned to an extended transcript. To prevent this, any extension will be truncated up to any downstream AS

### to call peaks, split each the cellranger-arc "atac_fragments.tsv.gz" output to its individual fragments, then call MACS on the merged set of fragment ends

bash scripts/macs2_cellranger_fragments_all.20231103.sh data/sc/cellranger_arc/exp031/outs/atac_fragments.tsv.gz,data/sc/cellranger_arc/exp032/outs/atac_fragments.tsv.gz,data/sc/cellranger_arc/exp042/outs/atac_fragments.tsv.gz,data/sc/cellranger_arc/exp043/outs/atac_fragments.tsv.gz scATAC_bulk_peaks

cut -f 1,2,3,4,5,6 scATAC_bulk_peaks/macs/resized_fragments.all_samples_peaks.narrowPeak > scATAC_bulk_peaks/macs/resized_fragments.all_samples_peaks.bed

### extend transcript annotation by 100/200/300/400/500bp (note, the "-100" contraction is not needed anymore, so I don't need to discard short transcripts)
mkdir species/elegans/gene_annotation/extended_no_ovlp
python scripts/gtf_3_utr_extension.20231108.py c_elegans.PRJNA13758.WS285.canonical_geneset.filtered.gtf 100 0,100,200,300,400 species/elegans/genome/elegans.chrom.sizes.txt c_elegans.PRJNA13758.WS285.canonical_geneset.filtered.gene_transcript_id.sorted.txt species/elegans/gene_annotation extended_no_ovlp


# strict extension
for extension in 100 200 300 400 500
do
